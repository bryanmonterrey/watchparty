// components/SmoothScrollManager.tsx
'use client';
import { useEffect, useRef } from 'react';

export default function SmoothScrollManager({ children }: { children: React.ReactNode }) {
  const wrapperRef = useRef<HTMLDivElement>(null);
  const scrollYRef = useRef(0);
  const velocityRef = useRef(0);
  const rafRef = useRef<number>();

  useEffect(() => {
    const wrapper = wrapperRef.current;
    if (!wrapper) return;

    let lastScroll = 0;
    let targetScroll = 0;

    const animate = () => {
      // Simple spring/momentum simulation (tune these numbers to match Cash App feel)
      velocityRef.current = (targetScroll - scrollYRef.current) * 0.12;
      scrollYRef.current += velocityRef.current;

      wrapper.style.transform = `translateY(${-scrollYRef.current}px)`;

      rafRef.current = requestAnimationFrame(animate);
    };

    const onScroll = () => {
      targetScroll = window.scrollY;
    };

    window.addEventListener('scroll', onScroll, { passive: true });
    rafRef.current = requestAnimationFrame(animate);

    return () => {
      window.removeEventListener('scroll', onScroll);
      if (rafRef.current) cancelAnimationFrame(rafRef.current);
    };
  }, []);

  return (
    <div 
      ref={wrapperRef}
      className="smooth-scroll-manager" 
      data-smooth-scroll-manager="true"
    >
      {children}
    </div>
  );
}

  const bgRef = useRef<HTMLDivElement>(null);
  const contentRef = useRef<HTMLDivElement>(null);

  useEffect(() => {
    const bg = bgRef.current;
    const content = contentRef.current;
    if (!bg || !content) return;

    let raf: number;

      const scroll = window.scrollY;
      const progress = Math.max(0, Math.min(1, scroll / window.innerHeight));
